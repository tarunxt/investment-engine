import React from "react";
import { createRoot } from "react-dom/client";
import { DashboardPageClient } from "../../app/console/dashboard/DashboardPageClient";
import { SidebarPolicyContext } from "@/hooks/useSidebarPolicy";

const scenario = new URLSearchParams(location.search).get("state");
const now = Date.now();
const stamp = new Date(now - 7200000).toISOString();
const older = new Date(now - 3 * 86400000).toISOString();
const india = { captured_at: stamp, holdings_market_value: 120000, available_margin: 8000 };
const us = { captured_at: stamp, current_value: 1000, wallet_balance: scenario === "mixed" ? null : 50 };
const fx = { value: 84, source: "Test", as_of: new Date(now).toISOString(), stale_after_seconds: 3600, status: "valid" };
const ids = ["india-portfolio", "us-portfolio"];
if (scenario === "enabled") ids.push("bullpen-ai-review");
window.reads = [];
window.fail = scenario === "cached";
if (window.fail) localStorage.setItem("investment-engine:dashboard-overview-cache:v3:user:9001", JSON.stringify({version:2,entries:{
  zerodhaOverview:{value:{latest:{...india,captured_at:older},history:[]}},
}}));
window.readSource = async source => {
  window.reads.push(source);
  if (scenario === "loading") return new Promise(() => {});
  if (window.fail) throw new Error("Fixture source offline");
  if (scenario === "empty") return {latest:null,history:[]};
  if (source === "zerodha") return {latest:india,history:[]};
  if (source === "indmoney") return {latest:us,history:scenario === "mixed" ? [{...us,captured_at:older,wallet_balance:0}] : []};
  return fx;
};

  <SidebarPolicyContext.Provider value={{visible_ids:ids,can_manage:true,revision:1}}>
    <DashboardPageClient initialFx={scenario === "reference" ? {...fx,status:"unavailable",value:null} : fx}/>
  </SidebarPolicyContext.Provider>
);
